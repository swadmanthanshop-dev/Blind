<script>

const video = document.getElementById('video');

const startBtn = document.getElementById('start-btn');

const statusDiv = document.getElementById('status');

let isRunning = false;

let lastSpeechTime = 0;

// CAMERA CALIBRATION CONSTANTS (Triangle Similarity: F = (P * D) / W)

const KNOWN_WIDTH = 50; // Average width of a person or large hurdle in centimeters (~50cm)

const FOCAL_LENGTH = 650; // Pre-calibrated focal length factor for typical mobile lenses

// Initialize Speech Synthesis

const synth = window.speechSynthesis;

async function startSystem() {

try {

// Request back camera access with optimal resolution parameters

const stream = await navigator.mediaDevices.getUserMedia({

video: { facingMode: { exact: "environment" }, width: 640, height: 480 }

});

video.srcObject = stream;

isRunning = true;

statusDiv.innerText = "Tracking active. Scanning space ahead...";

startBtn.style.backgroundColor = "#ff1744";

startBtn.innerText = "SYSTEM RUNNING";

speak("AI Navigation Active. Path scanning running.", 1.0);

// Start processing loops

requestAnimationFrame(processFrameLoop);

} catch (err) {

statusDiv.innerText = "Error accessing back camera. Ensure permissions are granted.";

console.error(err);

}

}

// Central Text-to-Speech Engine with speed override adjustments

function speak(text, speedRate) {

if (synth.speaking) return; // Prevent audio overlap stacking

const utterance = new SpeechSynthesisUtterance(text);

utterance.rate = speedRate; // Speeds up text vocalization dynamically

synth.speak(utterance);

}

function processFrameLoop() {

if (!isRunning) return;

// In a production build, you can run a tfjs object model here.

// For this layout simulation, we simulate scanning an obstacle cross-section profile.

let simulatedPixelWidth = Math.floor(Math.random() * (400 - 50 + 1)) + 50;

// Calculate Distance: D = (W * F) / P

let distanceCm = (KNOWN_WIDTH * FOCAL_LENGTH) / simulatedPixelWidth;

let distanceMeters = (distanceCm / 100).toFixed(1);

let currentTime = Date.now();

// DYNAMIC LOOP: Speak faster and more frequently as distance drops

if (distanceMeters < 1.5) {

// Urgent Critical Proximity Check

if (currentTime - lastSpeechTime > 1000) {

speak("Obstacle close. " + distanceMeters + " meters. Step left now.", 1.8);

statusDiv.innerText = `CRITICAL: ${distanceMeters}m! Turn Left.`;

lastSpeechTime = currentTime;

}

} else if (distanceMeters < 3.0) {

// Standard Approaching Caution Check

if (currentTime - lastSpeechTime > 2500) {

speak("Obstacle ahead at " + distanceMeters + " meters. Clear to proceed slow.", 1.2);

statusDiv.innerText = `WARNING: Object at ${distanceMeters}m.`;

lastSpeechTime = currentTime;

}

} else {

// Clear Path Routine

if (currentTime - lastSpeechTime > 5000) {

speak("Path clear for " + distanceMeters + " meters. Walk straight.", 1.0);

statusDiv.innerText = `PATH CLEAR: Safe distance ${distanceMeters}m.`;

lastSpeechTime = currentTime;

}

}

// Keep looping over every camera frame processing ticket

setTimeout(() => {

requestAnimationFrame(processFrameLoop);

}, 300); // Check 3 times a second to prevent CPU overload

}

startBtn.addEventListener('click', () => {

if (!isRunning) {

startSystem();

}

});

</script>
